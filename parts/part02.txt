const frHTML = (n, d) => `<span class="fr" data-n="${n}" data-d="${d}" role="img" aria-label="${fracWords(n, d)}"><span class="n">${n}</span><span class="d">${d}</span></span>`;
const optHTML = o => o.s ? `<span class="sym">${o.s}</span><span class="cap">${{ "<": "less than", ">": "greater than", "=": "equal to" }[o.s]}</span>` : o.w !== undefined ? `<span class="w">${o.w}</span>` : frHTML(o.n, o.d);
const rich = s => s.replace(/(\d+)\/(\d+)/g, (_, n, d) => frHTML(+n, +d));
function speechOf(el) {
  const c = el.cloneNode(true);
  $$(".say", c).forEach(b => b.remove());
  $$(".fr", c).forEach(f => f.replaceWith(" " + fracWords(+f.dataset.n, +f.dataset.d) + " "));
  $$(".sym", c).forEach(f => f.replaceWith(" " + (SYMWORD[f.textContent] || "") + " "));
  $$(".cap", c).forEach(f => f.remove());
  return c.textContent.replace(/(^|\s)([<>=])(?=\s|$)/g, (m, a, g) => a + SYMWORD[g]).replace(/[\u{1F300}-\u{1FAFF}\u{2600}-\u{27BF}\u{FE0F}]/gu, " ").replace(/\s+/g, " ").replace(/ ([.,!?:])/g, "$1").trim();
}

/* ---------- sound (synthesized, no files) ---------- */
const Sound = {
  on: LS.get("sound", true), ctx: null,
  init() { if (this.ctx || !this.on) return; try { const AC = window.AudioContext || window.webkitAudioContext; if (AC) this.ctx = new AC(); } catch (e) { this.ctx = null; } },
  tone(f, t0, dur, type = "sine", vol = .18, f2) {
    const c = this.ctx; if (!c) return;
    const o = c.createOscillator(), g = c.createGain(), t = c.currentTime + t0;
    o.type = type; o.frequency.setValueAtTime(f, t); if (f2) o.frequency.exponentialRampToValueAtTime(f2, t + dur);
    g.gain.setValueAtTime(.0001, t); g.gain.exponentialRampToValueAtTime(vol, t + .02); g.gain.exponentialRampToValueAtTime(.0001, t + dur);
    o.connect(g).connect(c.destination); o.start(t); o.stop(t + dur + .05);
  },
  play(name) {
    if (!this.on) return; this.init(); const c = this.ctx; if (!c) return;
    try {
      if (c.state === "suspended") c.resume();
      ({ tap: () => this.tone(660, 0, .08, "triangle", .12),
        good: () => [523, 659, 784].forEach((f, i) => this.tone(f, i * .09, .22, "triangle", .2)),
        oops: () => { this.tone(330, 0, .18, "sine", .12); this.tone(294, .16, .25, "sine", .1); },
        zap: () => { this.tone(1200, 0, .35, "sawtooth", .1, 120); this.tone(300, .05, .3, "square", .06, 900); },
        jump: () => this.tone(400, 0, .14, "triangle", .12, 800),
        badge: () => [523, 659, 784, 1047].forEach((f, i) => this.tone(f, i * .12, .35, "triangle", .2)),
        win: () => [392, 523, 659, 784, 659, 784, 1047].forEach((f, i) => this.tone(f, i * .14, .3, "square", .09)) })[name]();
    } catch (e) { }
  },
  toggle() { this.on = !this.on; LS.set("sound", this.on); if (this.on) this.play("tap"); syncSoundBtn(); }
};
function syncSoundBtn() { const b = $("#soundBtn"); b.firstChild.textContent = Sound.on ? "🔊" : "🔇"; $("#soundLbl").textContent = Sound.on ? "Sound" : "Muted"; b.setAttribute("aria-pressed", String(Sound.on)); b.setAttribute("aria-label", Sound.on ? "Sound on. Tap to mute." : "Sound off. Tap to turn on."); }

/* ---------- read aloud ---------- */
const Speech = {
  ok: typeof window !== "undefined" && "speechSynthesis" in window && typeof SpeechSynthesisUtterance !== "undefined",
  btn: null,
  say(text, btn) {
    if (!this.ok) { toast("Read-aloud is not available on this device."); return; }
    try {
      speechSynthesis.cancel(); if (this.btn) this.btn.classList.remove("speaking");
      const u = new SpeechSynthesisUtterance(text); u.rate = .88; u.pitch = 1.1; u.lang = "en-US";
      this.btn = btn || null; if (btn) btn.classList.add("speaking");
      u.onend = u.onerror = () => btn && btn.classList.remove("speaking");
      speechSynthesis.speak(u);
      setTimeout(() => btn && btn.classList.remove("speaking"), Math.min(15000, 600 + text.length * 80));
    } catch (e) { toast("Read-aloud is not available right now."); }
  },
  stop() { try { if (this.ok) speechSynthesis.cancel(); } catch (e) { } if (this.btn) this.btn.classList.remove("speaking"); }
};
function sayButton(target) {
  const b = H("button", { class: "say", "aria-label": "Read this out loud", title: "Read out loud" }, "🔈");
  b.onclick = e => { e.stopPropagation(); Speech.say(speechOf(target), b); };
  return b;
}
/* a hero face + speech bubble with a read-aloud button */
function talk(host, who, html) {
  host.innerHTML = ""; host.classList.add("talk");
  const face = H("div", { class: "face", "aria-hidden": "true" }, figure(who, true)); host.appendChild(face);
  const bub = H("div", { class: "bubble", "aria-live": "polite" }); host.appendChild(bub);
  setBubble(bub, html); return bub;
}
function setBubble(bub, html) { bub.innerHTML = `<div class="btxt">${rich(html)}</div>`; bub.prepend(sayButton($(".btxt", bub))); }

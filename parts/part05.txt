const Builder = (function builder() {
  const sel = $("#bMol"), svg = $("#bSvg"), solved = new Set(LS.get("solved", []));
  let m, s, hist = [], tool = "pair", stageK = 0, hintK = 0, ansK = 0;
  function load(id) { m = MOL[id]; s = emptyState(m); hist = []; stageK = 0; hintK = 0; ansK = 0; fb("", `Build <span class="chemf">${m.f}</span> (${m.name}). Start by counting: how many valence electrons are available?`); draw(); }
  function fb(cls, html) { const f = $("#bFeedback"); f.className = "feedback " + cls; f.innerHTML = html; }
  function push() { hist.push(norm(m, s)); if (hist.length > 100) hist.shift(); }
  function draw() {
    renderStructure(svg, m, s, { interactive: true, showFC: $("#bFC").checked && usedElectrons(s) > 0, octetColors: true, onBond, onAtom });
    const T = totalValence(m), U = usedElectrons(s), L = T - U;

    const cls = { ok: "okc", short: "warnc", over: "badc", expanded: "expc", "allowed-short": "" };
    $("#atomRows").innerHTML = m.atoms.map((a, i) => { const st = octetStatus(m, s, i), fc = formalCharge(m, s, i);
      return `<tr><td><b>${a}</b><sub class="muted">${i + 1}</sub></td><td>${atomElectrons(m, s, i)}</td><td class="${cls[st]}" style="${st === "allowed-short" ? "color:var(--accent3);font-weight:700" : ""}">${a === "H" && st === "ok" ? "✓ duet" : label[st]}</td><td>${$("#bFC").checked && U > 0 ? fcText(fc) : "–"}</td></tr>`; }).join("");
    $("#bSolved").textContent = `Solved ${solved.size} / ${MOLS.length}`;
    $("#bUndo").disabled = !hist.length;
  }
  function onBond(k) { push(); s.b[k] = (s.b[k] + 1) % 4; draw(); }
  function onAtom(i, e, remove) {
    push();
    const t = remove ? "erase" : tool;
    if (t === "pair") { if (s.lp[i] < 4) s.lp[i]++; else { hist.pop(); fb("info", "That's the most lone pairs an atom can hold here."); return; } }
    else if (t === "single") s.se[i] = s.se[i] ? 0 : 1;
    else { if (s.se[i]) s.se[i] = 0; else if (s.lp[i]) s.lp[i]--; else { hist.pop(); return; } }
    draw();
  }
  function hints() {
    const T = totalValence(m), E = m.edges.length, f = majorForms(m)[0].s, extra = f.b.reduce((a, c) => a + c, 0) - E;
    const h = [`Count first: ${breakdownText(m)}.`, centralText(m), `Single bonds: ${E} bond${E > 1 ? "s" : ""} × 2 = ${2 * E} electrons, leaving ${T - 2 * E} for lone pairs${extra ? " and multiple bonds" : ""}.`];
    if (extra) h.push(`After filling octets you'll come up short: the structure needs <b>${extra}</b> extra bond${extra > 1 ? "s" : ""} (${extra} double, or a triple if both go between the same two atoms). Convert outer lone pairs into bonds.`);

  }

      let msg = `✅ <b>${d.form.kind === "alt" ? "Accepted!" : "Correct!"}</b> ${fcList.length ? "Formal charges: " + fcList.join(", ") + "." : "Every formal charge is 0."}${shape}`;
      if (maj.length > 1 && d.form.kind === "major") msg += ` This is one of <b>${maj.length}</b> equivalent resonance forms. Tap 👀 Answer to cycle through them.`;
      if (d.form.note) msg += `<br><span class="small">${d.form.note}</span>`;
      else if (m.note && d.form.kind === "major") msg += `<br><span class="small">${m.note}</span>`;
      fb("ok", msg);
      if (!solved.has(m.id)) { solved.add(m.id); LS.set("solved", [...solved]); moleculeOptions(sel, undefined, solved); sel.value = m.id; Confetti.burst(90); }
      draw();
    } else fb(d.valid ? "info" : "no", (d.valid ? "🤔 " : "❌ ") + d.msg);
  };
  $("#bHint").onclick = () => { const h = hints(); fb("info", `💡 <b>Hint ${Math.min(hintK, h.length - 1) + 1}/${h.length}:</b> ${h[Math.min(hintK, h.length - 1)]}`); hintK++; };
  $("#bStep").onclick = () => { const st = stages(m); push(); const S0 = st[Math.min(stageK, st.length - 1)]; s = norm(m, S0.s); fb("info", `⏭ <b>Step ${Math.min(stageK, st.length - 1) + 1}/${st.length}: ${S0.title}.</b> ${S0.text}`); stageK++; draw(); };
  $("#bUndo").onclick = () => { if (hist.length) { s = hist.pop(); draw(); } };
  $("#bReset").onclick = () => { push(); s = emptyState(m); stageK = 0; draw(); fb("", "Board cleared."); };
  $("#bShow").onclick = () => { const forms = acceptedForms(m), f = forms[ansK % forms.length]; push(); s = norm(m, f.s); ansK++;
    fb("info", `👀 Accepted structure ${((ansK - 1) % forms.length) + 1} of ${forms.length}${f.kind === "alt" ? " (alternative)" : ""}. ${f.note || (forms.length > 1 ? "Resonance forms differ only in where electrons are placed." : "")}`); draw(); };
  $$("#bTool button").forEach(b => b.onclick = () => { $$("#bTool button").forEach(x => x.classList.toggle("on", x === b)); tool = b.dataset.t; });
  $("#bFC").onchange = draw;
  moleculeOptions(sel, undefined, solved); sel.value = "H2O"; sel.onchange = () => load(sel.value);
  load("H2O");
  return { load: id => { sel.value = id; load(id); }, get state() { return s; }, get mol() { return m; } };
})();

/* ---------- resonance ---------- */
(function resonance() {
  let m, forms, k = 0, playing = true, timer = 0;
  const svg = $("#resSvg");
  function hybrid() {
    const n = forms.length, orders = m.edges.map((_, e) => forms.reduce((a, f) => a + f.s.b[e], 0) / n), fc = m.atoms.map((_, i) => forms.reduce((a, f) => a + formalCharge(m, f.s, i), 0) / n);
    return { orders, fc };
  }
  function arrows(A, B, lay) {
    const g = S("g", {}, svg), P = m.pos, cx = P[m.center][0], cy = P[m.center][1];
    const defs = S("defs", {}, svg), mk = S("marker", { id: "ah" + svg.id, viewBox: "0 0 10 10", refX: 8, refY: 5, markerWidth: 7, markerHeight: 7, orient: "auto-start-reverse" }, defs);
    S("path", { d: "M 0 0 L 10 5 L 0 10 z", fill: "var(--accent3)" }, mk);
    const curve = (x1, y1, x2, y2, bend = 26) => {
      const mx = (x1 + x2) / 2, my = (y1 + y2) / 2, dx = x2 - x1, dy = y2 - y1, L = Math.hypot(dx, dy) || 1;
      let nx = -dy / L, ny = dx / L; if ((mx + nx - cx) ** 2 + (my + ny - cy) ** 2 < (mx - nx - cx) ** 2 + (my - ny - cy) ** 2) { nx = -nx; ny = -ny; }
      S("path", { d: `M ${x1} ${y1} Q ${mx + nx * bend} ${my + ny * bend} ${x2} ${y2}`, fill: "none", stroke: "var(--accent3)", "stroke-width": 2.4, "marker-end": `url(#ah${svg.id})`, class: "pop" }, g);
    };
    const mid = e => { const [a, c] = m.edges[e]; return [(P[a][0] + P[c][0]) / 2, (P[a][1] + P[c][1]) / 2]; };
    const inc = m.edges.map((_, e) => e).filter(e => B.b[e] > A.b[e]), dec = m.edges.map((_, e) => e).filter(e => B.b[e] < A.b[e]);
    if (m.ring) {
      const R = m.ring, ringEdge = (i, j) => m.edges.findIndex(([a, c]) => (a === R[i % 6] && c === R[j % 6]) || (a === R[j % 6] && c === R[i % 6]));
      for (let i = 0; i < 6; i++) { const e = ringEdge(i, i + 1); if (dec.includes(e)) { const e2 = ringEdge(i + 1, i + 2); const [x1, y1] = mid(e), [x2, y2] = mid(e2); curve(x1 + (cx - x1) * .18, y1 + (cy - y1) * .18, x2 + (cx - x2) * .18, y2 + (cy - y2) * .18, -14); } }
      return;
    }
    inc.forEach(e => { const [a, c] = m.edges[e], t = A.lp[a] > B.lp[a] ? a : c; const lp = lay.lpPos[t].filter(p => !p[2]); if (!lp.length) return;
      const [px, py] = lp[lp.length - 1], [mx, my] = mid(e); curve(px, py, mx + (P[t][0] - mx) * .15, my + (P[t][1] - my) * .15); });
    dec.forEach(e => { const [a, c] = m.edges[e], t = B.lp[a] > A.lp[a] ? a : c; const [mx, my] = mid(e), dx = P[t][0] - cx, dy = P[t][1] - cy, L = Math.hypot(dx, dy) || 1;
      curve(mx, my, P[t][0] + dx / L * 26 + dy / L * 14, P[t][1] + dy / L * 26 - dx / L * 14); });
  }
  function draw() {
    const f = forms[k % forms.length], next = forms[(k + 1) % forms.length];
    svg.style.opacity = 0;
    setTimeout(() => {
      const lay = renderStructure(svg, m, f.s, { showFC: true });
      if ($("#resArrows").checked && forms.length > 1) arrows(f.s, next.s, lay);
      svg.style.opacity = 1;
    }, 160);
    $("#resLbl").textContent = `Resonance form ${(k % forms.length) + 1} of ${forms.length}${$("#resArrows").checked ? " · arrows show how to reach the next form" : ""}`;
  }
  function load(id) {
    m = MOL[id]; forms = majorForms(m); k = 0;
    const h = hybrid();
    renderStructure($("#hybSvg"), m, { b: h.orders.map(() => 0), lp: m.atoms.map(() => 0), se: m.atoms.map(() => 0) }, { hybrid: h });
    const eqBond = h.orders.find(o => Math.abs(o - Math.round(o)) > 1e-9);
    const fcNon = [...new Set(h.fc.filter(x => Math.abs(x) > 1e-9 && Math.abs(x - Math.round(x)) > 1e-9).map(frac))];
    const extra = { O3: "Both O–O bonds are identical (128 pm), with bond order 1.5. The central O carries +1 and the end O atoms share the −1 (−½ each).",
      "NO3-": "All three N–O bonds are identical in length, with bond order 4/3; each O carries −2/3 on average. Trigonal planar, 120°.",
      "CO3^2-": "All three C–O bonds are identical in length, with bond order 4/3; each O carries −2/3. Trigonal planar, 120°.",
      SO2: "Octet resonance forms shown (S +1). Bond order 1.5 in this picture. The expanded-octet form O=S=O (all FC 0) is also commonly drawn; the builder accepts both.",
      C6H6: "All six C–C bonds are identical (139 pm), between C–C (154 pm) and C=C (134 pm). That's why benzene is often drawn as a hexagon with a circle inside." }[id];
    $("#resInfo").innerHTML = `<b>${forms.length} resonance forms.</b> ${eqBond ? `Hybrid bond order: <b>${frac(eqBond)}</b>. ` : ""}${fcNon.length ? `Average formal charge on equivalent atoms: <b>${fcNon.join(", ")}</b>. ` : ""}${extra}`;
    draw();
  }
  function tick() { if (playing) { k++; draw(); } }
  $$("#resSel button").forEach(b => b.onclick = () => { $$("#resSel button").forEach(x => x.classList.toggle("on", x === b)); load(b.dataset.m); });
  $("#resPlay").onclick = () => { playing = !playing; $("#resPlay").textContent = playing ? "⏸ Pause" : "▶ Play"; };
  $("#resNext").onclick = () => { k++; draw(); };
  $("#resArrows").onchange = draw;
  svg.style.transition = "opacity .16s";
  new IntersectionObserver(e => { clearInterval(timer); if (e[0].isIntersecting) timer = setInterval(tick, 3200); }).observe(svg);
  load("O3");
})();

const COLORS = { shade: SHADE, empty: PAPER, blue: "#2e86de", yellow: "#ffd23f", green: "#1fb36b" };
function sectorPath(cx, cy, r, a0, a1) {
  if (a1 - a0 >= 359.999) return `M ${cx - r} ${cy} a ${r} ${r} 0 1 0 ${2 * r} 0 a ${r} ${r} 0 1 0 ${-2 * r} 0`;
  const p = a => [cx + r * Math.cos((a - 90) * Math.PI / 180), cy + r * Math.sin((a - 90) * Math.PI / 180)];
  const [x0, y0] = p(a0), [x1, y1] = p(a1);
  return `M ${cx} ${cy} L ${x0.toFixed(2)} ${y0.toFixed(2)} A ${r} ${r} 0 ${a1 - a0 > 180 ? 1 : 0} 1 ${x1.toFixed(2)} ${y1.toFixed(2)} Z`;
}

function circleModel(d, on, o = {}) {
  const svg = S("svg", { viewBox: "0 0 220 220", class: "model", role: "img" }), set = new Set(on);
  const R = 96, cx = 110, cy = 110;
  if (o.pizza) S("circle", { cx, cy, r: R + 8, fill: "#e8a54b", stroke: INK, "stroke-width": 4 }, svg);

    case "circle": return circleModel(m.d, m.on, { pizza: m.pizza });

    case "grid": return gridModel(m.r, m.c, m.on);
    case "set": return setModel(m.icon, m.d, m.on);
    case "wholes": return wholesModel(m.count, m.d, m.on);
    case "cmp": return cmpModel(m.a, m.b);
    case "bars": return barsModel(m.list);
    case "line": return lineModel(m.range, m.d, { dot: m.dot }).svg;
  }
}

/* ---------- progress, badges ---------- */
const Progress = {
  badges: new Set(LS.get("badges", [])),
  has(i) { return this.badges.has(i); },
  earn(i) { const was = this.badges.has(i); this.badges.add(i); LS.set("badges", [...this.badges]); syncBadges(); return !was; },
  reset() { this.badges.clear(); LS.set("badges", []); syncBadges(); }
};
function syncBadges() { $("#badgeCount").textContent = Progress.badges.size; }

/* ---------- mini math typesetter ----------
   \f{num}{den}  fraction      \r{x}  square root
   ^{..} or ^c   superscript   _{..}  subscript
   '-' becomes a real minus sign; single-letter variables become italic.
   HTML tags pass through untouched. */
function tex(src) {
  let i = 0;
  function group() {               // read {...} or one char
    if (src[i] === "{") {
      let depth = 0, start = ++i;
      for (; i < src.length; i++) {
        if (src[i] === "{") depth++;
        else if (src[i] === "}") { if (depth === 0) break; depth--; }
      }
      const inner = src.slice(start, i); i++;
      return tex(inner);
    }
    if (src[i] === "\\") { const r = parseUntil(i, true); return r; }
    return fmtText(src[i++] || "");
  }
  function parseUntil(_s, single) {  // parse a single command at i
    let out = "";
    const c2 = src[i + 1];
    if (c2 === "f") { i += 2; const a = group(), b = group(); out = `<span class="fr"><span>${a}</span><span>${b}</span></span>`; }
    else if (c2 === "r") { i += 2; const a = group(); out = `<span class="rt"><span class="sq">√</span><span class="rad">${a}</span></span>`; }
    else { out = "\\"; i++; }
    return out;
  }
  let out = "", buf = "";
  const flush = () => { out += fmtText(buf); buf = ""; };
  while (i < src.length) {
    const c = src[i];
    if (c === "<") { flush(); const j = src.indexOf(">", i); out += src.slice(i, j + 1); i = j + 1; }
    else if (c === "&") { flush(); const j = src.indexOf(";", i); out += src.slice(i, j + 1); i = j + 1; }
    else if (c === "\\") { flush(); out += parseUntil(i); }
    else if (c === "^") { flush(); i++; out += `<sup>${group()}</sup>`; }
    else if (c === "_") { flush(); i++; out += `<sub>${group()}</sub>`; }
    else { buf += c; i++; }
  }
  flush();
  return out;
}
function fmtText(t) {
  return t.replace(/-/g, "−").replace(/(^|[^A-Za-z])([xyuθαβ])(?![A-Za-z])/g, "$1<i>$2</i>");
}

/* ---------- angle formatting ---------- */
function piFracTex(n, d, sym = "π") {       // value = (n/d)·π as tex
  if (n === 0) return "0";
  const g = gcd(n, d); n /= g; d /= g;
  if (d < 0) { n = -n; d = -d; }
  const sign = n < 0 ? "-" : ""; n = Math.abs(n);
  const num = (n === 1 ? "" : n) + sym;
  return d === 1 ? sign + num : `${sign}\\f{${num}}{${d}}`;
}
const degToRadTex = deg => Number.isInteger(deg) ? piFracTex(deg, 180) : null;
function piFracPlain(n, d) {   // for canvas/SVG text
  if (n === 0) return "0";
  const g = gcd(n, d); n /= g; d /= g;
  const sign = n < 0 ? "−" : ""; n = Math.abs(n);
  const num = (n === 1 ? "" : n) + "π";
  return d === 1 ? sign + num : `${sign}${num}/${d}`;
}
const fmtNum = (v, dp = 4) => {
  if (!isFinite(v)) return "undefined";
  if (Math.abs(v) < 5e-10) v = 0;
  let s = (+v.toFixed(dp)).toString();
  return s.replace("-", "−");
};

function refAngleDeg(d) { d = ((d % 360) + 360) % 360; return d <= 90 ? d : d <= 180 ? 180 - d : d <= 270 ? d - 180 : 360 - d; }
function exactTrig(deg) {         // {cos, sin, tan} as tex; tan null = undefined
  deg = ((deg % 360) + 360) % 360;
  const r = refAngleDeg(deg), m = MAG[r];
  if (!m) return null;
  const c = Math.cos(deg * PI / 180), s = Math.sin(deg * PI / 180);
  const sx = Math.abs(c) < 1e-9 ? 0 : Math.sign(c), sy = Math.abs(s) < 1e-9 ? 0 : Math.sign(s);
  const sg = (sign, v) => (v === "0" ? "0" : (sign < 0 ? "-" : "") + v);
  const sgP = (sign, v) => (v === "0" ? "0" : (sign < 0 ? "−" : "") + v);
  return {
    cos: sg(sx, m.cos), sin: sg(sy, m.sin),
    tan: m.tan === null ? null : sg(sx * sy, m.tan),
    plain: `(${sgP(sx, m.cp)}, ${sgP(sy, m.sp)})`
  };
}

/* ---------- safe expression parser (no eval) ---------- */
const FUNCS = {
  sin: Math.sin, cos: Math.cos, tan: Math.tan,
  csc: x => 1 / Math.sin(x), sec: x => 1 / Math.cos(x), cot: x => Math.cos(x) / Math.sin(x),
  sqrt: Math.sqrt, abs: Math.abs
};
const WORDS = ["theta", "sqrt", "sin", "cos", "tan", "csc", "sec", "cot", "abs", "pi", "x"];
function tokenize(str) {
  let s = String(str).toLowerCase()
    .replace(/[−–—]/g, "-").replace(/[×·⋅∙]/g, "*").replace(/÷/g, "/")
    .replace(/π/g, "pi").replace(/θ/g, "x").replace(/²/g, "^2").replace(/³/g, "^3")
    .replace(/√/g, "sqrt").replace(/[\[{]/g, "(").replace(/[\]}]/g, ")");
  const toks = [];
  let i = 0;
  while (i < s.length) {
    const c = s[i];
    if (/\s/.test(c)) { i++; continue; }
    if (/[0-9.]/.test(c)) {
      const m = s.slice(i).match(/^(\d+\.?\d*|\.\d+)/);
      if (!m) throw new Error(`I can't read the number near "${s.slice(i, i + 4)}"`);
      toks.push({ t: "num", v: parseFloat(m[1]), s: m[1] }); i += m[1].length; continue;
    }
    if (/[a-z]/.test(c)) {
      let w = s.slice(i).match(/^[a-z]+/)[0], j = 0;
      while (j < w.length) {
        const name = WORDS.find(n => w.startsWith(n, j));
        if (!name) throw new Error(`I don't recognize "${w.slice(j)}". Try sin, cos, tan, csc, sec, cot, sqrt, pi, x`);
        toks.push(name === "pi" ? { t: "pi" } : (name === "x" || name === "theta") ? { t: "x" } : { t: "fn", v: name });
        j += name.length;
      }
      i += w.length; continue;
    }
    if ("+-*/^()".includes(c)) { toks.push({ t: c }); i++; continue; }
    throw new Error(`I don't understand the symbol "${c}"`);
  }
  return toks;
}
function parseExpr(str, opts = {}) {
  const toks = tokenize(str);
  if (!toks.length) throw new Error("Type an expression");
  let p = 0;
  const peek = () => toks[p], next = () => toks[p++];
  const is = t => toks[p] && toks[p].t === t;
  const expect = t => { if (!is(t)) throw new Error(t === ")" ? "Missing a closing parenthesis )" : `Expected "${t}"`); p++; };
  const startsFactor = () => { const k = peek(); return k && (k.t === "num" || k.t === "x" || k.t === "pi" || k.t === "fn" || k.t === "("); };
  const startsArgAtom = () => { const k = peek(); return k && (k.t === "num" || k.t === "x" || k.t === "pi" || k.t === "("); };
  function expr() {
    let a = term();
    while (is("+") || is("-")) { const op = next().t; a = { t: "bin", op, a, b: term() }; }
    return a;
  }
  function term() {
    let a = unary();
    for (;;) {
      if (is("*") || is("/")) { const op = next().t; a = { t: "bin", op, a, b: unary() }; }
      else if (startsFactor()) a = { t: "bin", op: "*", a, b: power(), imp: true };
      else break;
    }
    return a;
  }
  function unary() {
    if (is("-")) { next(); return { t: "neg", a: unary() }; }
    if (is("+")) { next(); return unary(); }
    return power();
  }
  function power() {
    const base = primary();
    if (is("^")) { next(); return { t: "bin", op: "^", a: base, b: unary() }; }
    return base;
  }
  function atom() {
    const k = next();
    if (!k) throw new Error("The expression ends too early");
    if (k.t === "num") return { t: "num", v: k.v, s: k.s };
    if (k.t === "pi") return { t: "pi" };
    if (k.t === "x") { if (opts.noVar) throw new Error("Give a number here, not x"); return { t: "x" }; }
    if (k.t === "(") { const e = expr(); expect(")"); return { t: "par", a: e }; }
    p--; return null;
  }
  function primary() {
    const k = peek();
    if (!k) throw new Error("The expression ends too early");
    if (k.t === "fn") {
      next();
      let ex = null;
      if (is("^")) {           // sin^2 x
        next();
        if (is("-")) { next(); const n = next(); if (!n || n.t !== "num") throw new Error("Put a number after ^"); ex = { t: "neg", a: { t: "num", v: n.v, s: n.s } }; }
        else { ex = atom(); if (!ex) throw new Error("Put a number after ^"); }
      }
      let arg;
      if (is("(")) { next(); arg = expr(); expect(")"); }
      else {                    // sin 2x, sin x  (argument without parentheses)
        let neg = false;
        if (is("-")) { next(); neg = true; }
        if (!startsArgAtom()) throw new Error(`${k.v} needs an input, like ${k.v}(x)`);
        arg = argAtom();
        while (startsArgAtom()) arg = { t: "bin", op: "*", a: arg, b: argAtom(), imp: true };
        if (neg) arg = { t: "neg", a: arg };
      }
      const f = { t: "fn", f: k.v, a: arg };
      return ex ? { t: "bin", op: "^", a: f, b: ex, fnpow: true } : f;
    }
    const a = atom();
    if (!a) throw new Error(k.t === ")" ? "There's an extra )" : `Something is missing before "${k.t}"`);
    return a;
  }
  function argAtom() {
    const a = atom();
    if (is("^")) { next(); return { t: "bin", op: "^", a, b: unary() }; }
    return a;
  }
  const tree = expr();
  if (p < toks.length) throw new Error(toks[p].t === ")" ? "There's an extra )" : "I couldn't understand the end of that");
  return tree;
}
function compile(n) {
  switch (n.t) {
    case "num": { const v = n.v; return () => v; }
    case "pi": return () => PI;
    case "x": return x => x;
    case "par": return compile(n.a);
    case "neg": { const a = compile(n.a); return x => -a(x); }
    case "fn": { const f = FUNCS[n.f], a = compile(n.a); return x => f(a(x)); }
    case "bin": {
      const a = compile(n.a), b = compile(n.b);
      switch (n.op) {
        case "+": return x => a(x) + b(x);
        case "-": return x => a(x) - b(x);
        case "*": return x => a(x) * b(x);
        case "/": return x => a(x) / b(x);
        case "^": return x => Math.pow(a(x), b(x));
      }
    }
  }
  throw new Error("bad node");
}
/* pretty-print a parsed tree back into tex (so she can see how it was read) */
function toTex(n) {
  const P = { num: 9, x: 9, pi: 9, par: 9, fn: 8, neg: 3 };
  const prec = k => k.t === "bin" ? ({ "+": 1, "-": 1, "*": 2, "/": 2, "^": 4 })[k.op] : P[k.t];
  const wrap = (k, minP) => prec(k) < minP ? "(" + toTex(k) + ")" : toTex(k);
  const fnArg = a => (a.t === "x" || a.t === "num" || a.t === "pi" || (a.t === "bin" && a.imp && a.a.t === "num" && (a.b.t === "x" || a.b.t === "pi"))) ? " " + toTex(a) : (a.t === "par" ? toTex(a) : "(" + toTex(a) + ")");
  switch (n.t) {
    case "num": return n.s;
    case "x": return "x";
    case "pi": return "π";
    case "par": return "(" + toTex(n.a) + ")";
    case "neg": return "-" + wrap(n.a, 3);
    case "fn":
      if (n.f === "sqrt") return `\\r{${n.a.t === "par" ? toTex(n.a.a) : toTex(n.a)}}`;
      if (n.f === "abs") return `|${n.a.t === "par" ? toTex(n.a.a) : toTex(n.a)}|`;
      return n.f + fnArg(n.a);
    case "bin": {
      const { op, a, b } = n;
      if (op === "/") return `\\f{${a.t === "par" ? toTex(a.a) : toTex(a)}}{${b.t === "par" ? toTex(b.a) : toTex(b)}}`;
      if (op === "^") {
        const e = b.t === "par" ? toTex(b.a) : toTex(b);
        if (a.t === "fn" && !["sqrt", "abs"].includes(a.f)) return `${a.f}^{${e}}${fnArg(a.a)}`;
        return `${wrap(a, 5)}^{${e}}`;
      }
      if (op === "*") {
        const L = wrap(a, 2), R = wrap(b, 2.5);
        const needDot = /^[0-9.]/.test(R) || (!n.imp);
        return L + (needDot ? " · " : (b.t === "x" || b.t === "pi" || b.t === "par") && a.t === "num" ? "" : " ") + R;
      }
      return `${wrap(a, 1)} ${op === "+" ? "+" : "-"} ${wrap(b, 2)}`;
    }
  }
  return "?";
}
function makeFn(str, opts) { const tree = parseExpr(str, opts); return { tree, f: compile(tree), tex: toTex(tree) }; }

function prepCanvas(cv, aspect) {
  const w = Math.max(240, cv.clientWidth || cv.parentElement.clientWidth);
  const h = Math.round(w / aspect);
  cv.style.height = h + "px";
  const dpr = Math.min(window.devicePixelRatio || 1, 2.5);
  if (cv.width !== Math.round(w * dpr) || cv.height !== Math.round(h * dpr)) { cv.width = Math.round(w * dpr); cv.height = Math.round(h * dpr); }
  const ctx = cv.getContext("2d");
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  ctx.clearRect(0, 0, w, h);
  return { ctx, w, h };
}
class Plot {
  constructor(ctx, box, xr, yr) {
    Object.assign(this, { ctx, box, x0: xr[0], x1: xr[1], y0: yr[0], y1: yr[1] });
  }
  X(x) { const b = this.box; return b.l + (x - this.x0) / (this.x1 - this.x0) * (b.r - b.l); }
  Y(y) { const b = this.box; return b.b - (y - this.y0) / (this.y1 - this.y0) * (b.b - b.t); }
  grid({ piStep = 2, yStep = 1, labels = true } = {}) {
    const c = this.ctx, b = this.box;
    c.save();
    c.lineWidth = 1; c.strokeStyle = cssVar("--grid"); c.fillStyle = cssVar("--muted");
    c.font = "12px system-ui, sans-serif";
    const k0 = Math.ceil(this.x0 / (PI / piStep)), k1 = Math.floor(this.x1 / (PI / piStep));
    for (let k = k0; k <= k1; k++) {
      const X = this.X(k * PI / piStep);
      c.beginPath(); c.moveTo(X, b.t); c.lineTo(X, b.b); c.stroke();
    }
    for (let y = Math.ceil(this.y0 / yStep) * yStep; y <= this.y1 + 1e-9; y += yStep) {
      const Y = this.Y(y); c.beginPath(); c.moveTo(b.l, Y); c.lineTo(b.r, Y); c.stroke();
    }
    c.strokeStyle = cssVar("--axis"); c.lineWidth = 1.5;
    if (this.y0 <= 0 && this.y1 >= 0) { c.beginPath(); c.moveTo(b.l, this.Y(0)); c.lineTo(b.r, this.Y(0)); c.stroke(); }
    if (this.x0 <= 0 && this.x1 >= 0) { c.beginPath(); c.moveTo(this.X(0), b.t); c.lineTo(this.X(0), b.b); c.stroke(); }
    if (labels) {
      c.textAlign = "center"; c.textBaseline = "top";
      const Y0 = clamp(this.Y(0) + 4, b.t, b.b - 14);
      const skip = (b.r - b.l) / (k1 - k0 + 1) < 34 ? 2 : 1;
      for (let k = k0; k <= k1; k++) { if (k === 0 || k % skip) continue; c.fillText(piFracPlain(k, piStep), this.X(k * PI / piStep), Y0); }
      c.textAlign = "right"; c.textBaseline = "middle";
      const X0 = clamp(this.X(0) - 5, b.l + 16, b.r);
      for (let y = Math.ceil(this.y0 / yStep) * yStep; y <= this.y1 + 1e-9; y += yStep) { if (Math.abs(y) < 1e-9) continue; c.fillText(fmtNum(y, 2), X0, this.Y(y)); }
    }
    c.restore();
  }
  fn(f, { color, width = 2.5, dash = null, n = 900, breakJump = true, from = this.x0, to = this.x1 } = {}) {
    const c = this.ctx, b = this.box, span = this.y1 - this.y0;
    c.save();
    c.beginPath(); c.rect(b.l, b.t - 2, b.r - b.l, b.b - b.t + 4); c.clip();
    c.strokeStyle = color; c.lineWidth = width; c.lineJoin = "round"; c.lineCap = "round";
    if (dash) c.setLineDash(dash);
    c.beginPath();
    let pen = false, prev = null;
    for (let i = 0; i <= n; i++) {
      const x = from + (to - from) * i / n, y = f(x);
      const ok = isFinite(y) && Math.abs(y) < 1e6;
      if (!ok) { pen = false; prev = null; continue; }
      if (pen && breakJump && prev !== null && Math.abs(y - prev) > span * 1.2 && Math.sign(y) !== Math.sign(prev)) pen = false;
      const Y = clamp(this.Y(y), b.t - span * 50, b.b + span * 50);
      if (pen) c.lineTo(this.X(x), Y); else c.moveTo(this.X(x), Y);
      pen = true; prev = y;
    }
    c.stroke(); c.restore();
  }
}

function drawSpecialLabels(g, { r = 1.1, fs = 0.07, deg = true } = {}) {
  SPECIAL.forEach(d => {
    const a = d * PI / 180, c = Math.cos(a), s = Math.sin(a), ex = exactTrig(d);

    if (d === 90) y -= fs * 1.4; if (d === 270) y += fs * 1.2;

    S("tspan", { x, dy: -fs * 0.15, "font-weight": 700, text: rad + (deg ? `  ${d}°` : ""), style: "fill:var(--accent)" }, t);
    S("tspan", { x, dy: fs * 1.15, text: ex.plain, style: "fill:var(--text)" }, t);
  });
}

/* ---------- nav chips, progress bar, visited ---------- */
(function nav() {
  const secs = $$("main section"), box = $("#navchips"), visited = new Set(LS.get("visited", []));
  const chips = secs.map((s, i) => {
    const a = document.createElement("a");
    a.className = "chip"; a.href = "#" + s.id; a.innerHTML = `<span class="n">${visited.has(s.id) ? "✓" : i + 1}</span>${s.dataset.title}`;
    if (visited.has(s.id)) a.classList.add("visited");
    box.appendChild(a); return a;
  });
  let current = -1;
  function update() {
    const y = scrollY, H = document.documentElement.scrollHeight - innerHeight;
    $("#progbar").style.width = (H > 0 ? y / H * 100 : 0) + "%";
    let idx = 0;
    secs.forEach((s, i) => { if (s.getBoundingClientRect().top < innerHeight * 0.35) idx = i; });
    if (idx !== current) {
      current = idx;
      chips.forEach((c, i) => c.classList.toggle("active", i === idx));
      const c = chips[idx]; box.scrollTo({ left: c.offsetLeft - box.clientWidth / 2 + c.clientWidth / 2, behavior: "smooth" });
      const id = secs[idx].id;
      if (!visited.has(id)) { visited.add(id); LS.set("visited", [...visited]); c.classList.add("visited"); c.querySelector(".n").textContent = "✓"; }
    }
  }
  addEventListener("scroll", update, { passive: true }); addEventListener("resize", update); update();
})();

  function frame(ts) {

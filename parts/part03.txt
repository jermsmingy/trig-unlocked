function totalValence(m) { return m.atoms.reduce((t, a) => t + EL[a].ve, 0) - m.q; }
function valenceBreakdown(m) {
  const cnt = {}; m.atoms.forEach(a => cnt[a] = (cnt[a] || 0) + 1);
  return { parts: Object.entries(cnt).map(([el, n]) => ({ el, n, ve: EL[el].ve, sub: n * EL[el].ve })), q: m.q, total: totalValence(m) };

function bondOrderSum(m, s, i) { let t = 0; m.edges.forEach(([a, c], k) => { if (a === i || c === i) t += s.b[k]; }); return t; }
function usedElectrons(s) { return 2 * s.b.reduce((a, c) => a + c, 0) + 2 * s.lp.reduce((a, c) => a + c, 0) + s.se.reduce((a, c) => a + c, 0); }
function atomElectrons(m, s, i) { return 2 * bondOrderSum(m, s, i) + 2 * s.lp[i] + s.se[i]; }
function formalCharge(m, s, i) { return EL[m.atoms[i]].ve - (2 * s.lp[i] + s.se[i]) - bondOrderSum(m, s, i); }
/* octet status: 'ok' | 'short' | 'over' | 'expanded' | 'allowed-short' */
function octetStatus(m, s, i) {
  const e = atomElectrons(m, s, i), el = m.atoms[i], P = EL[el].period;
  if (el === "H") return e === 2 ? "ok" : e < 2 ? "short" : "over";
  if (P === 2) {
    if (e > 8) return "over";
    if (e === 8) return "ok";
    if (el === "B" && e === 6) return "allowed-short";
    if (el === "Be" && e === 4) return "allowed-short";
    if (totalValence(m) % 2 === 1 && e === 7) return "allowed-short";
    return "short";
  }
  if (e === 8) return "ok";
  if (e > 8) return m.edges.filter(x => x.includes(i)).length <= 1 ? "over" : "expanded"; // only central atoms expand
  if (totalValence(m) % 2 === 1 && e === 7) return "allowed-short";
  return "short";
}

function permutations(arr) { if (arr.length <= 1) return [arr.slice()]; const out = []; arr.forEach((x, i) => permutations(arr.slice(0, i).concat(arr.slice(i + 1))).forEach(p => out.push([x].concat(p)))); return out; }
function edgeIndex(m, a, c) { return m.edges.findIndex(([x, y]) => (x === a && y === c) || (x === c && y === a)); }
function applyPerm(m, s, map) {
  const out = { b: rep(0, m.edges.length), lp: rep(0, m.atoms.length), se: rep(0, m.atoms.length) };
  s.lp.forEach((v, i) => out.lp[map[i]] = v); s.se.forEach((v, i) => out.se[map[i]] = v);
  m.edges.forEach(([a, c], k) => { const j = edgeIndex(m, map[a], map[c]); if (j < 0) throw new Error("bad symmetry group in " + m.id); out.b[j] = s.b[k]; });
  return out;
}
const _ansCache = {};
function acceptedForms(m) {
  if (_ansCache[m.id]) return _ansCache[m.id];
  let maps = [m.atoms.map((_, i) => i)];
  for (const g of m.eq) {
    const next = [];
    for (const base of maps) for (const p of permutations(g)) { const mp = base.slice(); g.forEach((orig, k) => mp[orig] = p[k]); next.push(mp); }
    maps = next;
  }
  const forms = [], seen = new Set();
  m.ans.forEach((a, ai) => {
    const s0 = norm(m, a);
    for (const mp of maps) { const s = applyPerm(m, s0, mp), k = stateKey(s); if (!seen.has(k)) { seen.add(k); forms.push({ s, key: k, kind: a.kind || "major", note: a.note || "", ai }); } }
  });
  return (_ansCache[m.id] = forms);
}
const majorForms = m => acceptedForms(m).filter(f => f.kind === "major");

function domainsAt(m, s, i) {
  if (m.atoms.length === 2) return { bonded: 1, lone: s.lp[i], domains: 0, info: { eg: "—", shape: "linear (diatomic)", angle: "—" } };
  const nb = m.edges.filter(([a, c], k) => (a === i || c === i) && s.b[k] > 0).length;
  const lone = s.lp[i] + (s.se[i] ? 1 : 0);   // a single electron counts as a (small) domain
  let info = VSEPR[(nb + lone) + "," + lone] || null;
  if (info && i === m.center && m.angle) info = Object.assign({}, info, { angle: m.angle });
  return { bonded: nb, lone, domains: nb + lone, info };
}

function centralAtoms(m) {
  const deg = m.atoms.map((_, i) => m.edges.filter(e => e.includes(i)).length);
  let c = m.atoms.map((_, i) => i).filter(i => deg[i] > 1);
  if (!c.length) { // diatomic: less electronegative atom plays "central"
    const [a, b] = [0, 1]; c = [(EL[m.atoms[a]].en || 9) <= (EL[m.atoms[b]].en || 9) ? a : b];
  }
  return c;
}
function stages(m) {
  const T = totalValence(m), out = [];
  const s1 = emptyState(m); s1.b = rep(1, m.edges.length);
  let left = T - usedElectrons(s1);
  out.push({ s: norm(m, s1), title: "Connect with single bonds", text: `${m.edges.length} single bond${m.edges.length > 1 ? "s" : ""} use ${2 * m.edges.length} electrons, leaving <b>${left}</b>.` });
  const s2 = norm(m, s1), cen = centralAtoms(m);
  const outer = m.atoms.map((_, i) => i).filter(i => !cen.includes(i) && m.atoms[i] !== "H");
  for (const i of outer) { while (atomElectrons(m, s2, i) < 8 && left >= 2) { s2.lp[i]++; left -= 2; } }
  out.push({ s: s2, title: "Complete the outer atoms' octets", text: outer.length ? `Give each outer atom lone pairs until it has 8 electrons (H needs only 2). Electrons left: <b>${left}</b>.` : "No outer atoms other than H, which are already happy with 2." });
  const s3 = norm(m, s2);
  while (left >= 2) { const i = cen.slice().sort((a, b) => atomElectrons(m, s3, a) - atomElectrons(m, s3, b))[0]; s3.lp[i]++; left -= 2; }
  if (left === 1) { const i = cen.slice().sort((a, b) => atomElectrons(m, s3, a) - atomElectrons(m, s3, b))[0]; s3.se[i] = 1; left = 0; }
  const shortC = cen.filter(i => ["short"].includes(octetStatus(m, s3, i)));
  out.push({ s: s3, title: "Put leftovers on the central atom", text: `Any remaining electrons go on the central atom${cen.length > 1 ? "s" : ""}. ${shortC.length ? `Now ${shortC.map(i => m.atoms[i]).join(", ")} still lack${shortC.length === 1 ? "s" : ""} an octet.` : "Every atom is satisfied."}` });
  const fin = majorForms(m)[0].s, extra = fin.b.reduce((a, c) => a + c, 0) - m.edges.length;
  if (extra && !shortC.length) {
    const fc3 = cen.map(i => `${m.atoms[i]} ${formalCharge(m, s3, i) > 0 ? "+" : ""}${formalCharge(m, s3, i)}`).join(", ");
    out.push({ s: norm(m, fin), title: "Lower formal charges (expanded octet)", text: `Every atom already has an octet, but the formal charges are large (${fc3}). A period-3 central atom may hold more than 8 electrons, so turning ${extra} outer lone pair${extra > 1 ? "s" : ""} into double bonds lowers the formal charges. Many textbooks call this the best structure; the octet form is also accepted (see the formal-charge section).` });
    return out;
  }
  out.push({ s: norm(m, fin), title: extra ? "Form multiple bonds" : "Check formal charges", text: extra ? `Turn ${extra} outer lone pair${extra > 1 ? "s" : ""} into bonding pair${extra > 1 ? "s" : ""} (${extra > 1 ? "multiple bonds" : "a double bond"}) so every atom has an octet, then check formal charges.` : "No multiple bonds needed. Check the formal charges: they should be as close to zero as possible and add up to the overall charge." });
  return out;
}

/* ---------- diagnose a student structure ---------- */
function diagnose(m, s) {
  s = norm(m, s);
  const T = totalValence(m), U = usedElectrons(s), name = i => `${m.atoms[i]}${m.atoms.filter(a => a === m.atoms[i]).length > 1 ? "" : ""}`;
  const hit = matchAnswer(m, s);
  if (hit) return { ok: true, form: hit };
  const missing = m.edges.findIndex((_, k) => s.b[k] === 0);
  if (missing >= 0) { const [a, c] = m.edges[missing]; return { ok: false, msg: `Every atom has to be connected. The ${m.atoms[a]}–${m.atoms[c]} link still has no bond (tap the dashed line).` }; }
  if (U !== T) return { ok: false, msg: U < T ? `You've placed ${U} electrons, but ${m.f} has <b>${T}</b>. Add ${T - U} more.` : `You've placed ${U} electrons, but ${m.f} only has <b>${T}</b>. Remove ${U - T}.` };
  for (let i = 0; i < m.atoms.length; i++) { const st = octetStatus(m, s, i), e = atomElectrons(m, s, i);
    if (m.atoms[i] === "H" && st !== "ok") return { ok: false, msg: `Hydrogen holds exactly 2 electrons (one single bond, no lone pairs). One H has ${e}.` };
    if (st === "over") return { ok: false, msg: EL[m.atoms[i]].period === 2 ? `${m.atoms[i]} has ${e} electrons. Period-2 atoms can't exceed an octet (only 4 orbitals: 2s + 2p).` : `An outer ${m.atoms[i]} has ${e} electrons. Outer (terminal) atoms keep a plain octet; only a central atom from period 3 or below can hold more than 8.` }; }
  for (let i = 0; i < m.atoms.length; i++) { if (octetStatus(m, s, i) === "short") { const e = atomElectrons(m, s, i);
    return { ok: false, msg: `${m.atoms[i]} has only ${e} electrons. ${m.atoms[i] === "B" || m.atoms[i] === "Be" ? "" : "Try turning a lone pair on a neighbor into a shared (multiple) bond."}` }; } }
  const fcs = s.lp.map((_, i) => formalCharge(m, s, i)), sum = fcs.reduce((a, c) => a + Math.abs(c), 0);
  const best = Math.min(...majorForms(m).map(f => f.s.lp.reduce((a, _, i) => a + Math.abs(formalCharge(m, f.s, i)), 0)));
  if (sum > best) return { ok: false, valid: true, msg: `This is a <b>valid</b> Lewis structure (right electron count, octets satisfied), but not the best one: your formal charges add up to Σ|FC| = ${sum}, and a structure with Σ|FC| = ${best} exists. Minimize formal charges!` };

/* ---------- Mission 4: number line leap ---------- */
function mission4() {
  const F = missionFrame(4, "Help Captain Fraction leap! Tap a spot on the line, or drag the hero.", M4.length);
  let i = 0, done = 0;
  function show() {
    const T = M4[i], rg = T.range; let cur = 0, busy = false, solved = false;

    const counter = S("text", { x: LINE.w / 2, y: G.top + 34 * B, "text-anchor": "middle", "font-size": 28 * B, "font-weight": 900, "font-family": "Arial Black,system-ui", fill: "#ef3e36", text: "" }, svg);
    const fb = $(".feedback", F.stage), ans = $(".answerline", F.stage), nextRow = $(".nextrow", F.stage);
    const step = (LINE.x1 - LINE.x0) / rg / T.d;
    for (let k = 0; k <= T.d * rg; k++) { const zx = Math.max(0, tickX(k, T.d, rg) - step / 2), zw = Math.min(LINE.w, tickX(k, T.d, rg) + step / 2) - zx; const z = S("rect", { class: "tickzone", x: zx, y: G.top + 20, width: zw, height: G.hgt - 20, "data-k": k, role: "button", "aria-label": `spot ${k}` }, svg); z.addEventListener("click", () => jumpTo(k)); }
    svg.appendChild(hero);
    function hopAnim(from, to, done) {
      const dir = to > from ? 1 : -1; let k = from;
      (function hop() {
        if (k === to) { done(); return; }
        const a = tickX(k, T.d, rg), b = tickX(k + dir, T.d, rg), t0 = performance.now(); Sound.play("jump");
        (function fr(t) { const p = Math.min(1, (t - t0) / 260); hero.place(a + (b - a) * p, hero.baseY - Math.sin(p * Math.PI) * 40 * B); if (p < 1) requestAnimationFrame(fr); else { k += dir; counter.textContent = k === 0 ? "0" : `${k}/${T.d}`; hop(); } })(t0);
      })();
    }
    function jumpTo(k) {
      if (busy || solved) return; busy = true;
      hopAnim(cur, k, () => { cur = k; busy = false; check(k); });
    }
    function check(k) {
      if (k === T.k) {
        solved = true; done++; F.setStars(done); Sound.play("good"); burst();

        S("line", { x1: -16, y1: 58, x2: 16, y2: 58, stroke: INK, "stroke-width": 3 }, lab);

        const nb = H("button", { class: "btn go big" }, i + 1 < M4.length ? "Next jump ▶" : "Finish! ⭐"); nextRow.appendChild(nb);
        nb.onclick = () => { if (i + 1 < M4.length) { i++; show(); } else awardBadge(4); };
      } else {
        Sound.play("oops");
        if (!$(".flabel", svg)) { const { svg: s2 } = lineModel(rg, T.d, { labels: true, big }); $$(".flabel", s2).forEach(g => svg.insertBefore(g, hero)); }
        fbSet(fb, "hint", `💡 You are on ${k === 0 ? "0" : k + "/" + T.d}. Count jumps from 0. You need ${T.k} jump${T.k === 1 ? "" : "s"} of 1/${T.d}.`);
      }
    }
    // drag the hero
    let drag = false;
    const toSvgX = e => { const pt = svg.createSVGPoint(); pt.x = e.clientX; pt.y = e.clientY; return pt.matrixTransform(svg.getScreenCTM().inverse()).x; };
    hero.style.cursor = "grab"; hero.style.touchAction = "none";
    hero.addEventListener("pointerdown", e => { if (busy || solved) return; drag = true; hero.setPointerCapture(e.pointerId); e.preventDefault(); });
    hero.addEventListener("pointermove", e => { if (!drag) return; const x = Math.max(LINE.x0, Math.min(LINE.x1, toSvgX(e))); hero.place(x); const kk = nearestTick(x, T.d, rg); counter.textContent = kk === 0 ? "0" : `${kk}/${T.d}`; });
    const end = e => { if (!drag) return; drag = false; const k = nearestTick(Math.max(LINE.x0, Math.min(LINE.x1, toSvgX(e))), T.d, rg); hero.place(tickX(k, T.d, rg)); cur = k; Sound.play("jump"); check(k); };
    hero.addEventListener("pointerup", end); hero.addEventListener("pointercancel", end);
  }
  show();
}

      if (st && st.ref) { const x = X0 + W * shade[st.ref] / st.ref; S("line", { x1: x, y1: 0, x2: x, y2: WALL.length * (RH + GAP) + 4, stroke: "#ef3e36", "stroke-width": 4, "stroke-dasharray": "10 7", "pointer-events": "none" }, svg); }

          [a, c].forEach(x => { x.classList.remove("nope"); void x.offsetWidth; x.classList.add("nope"); });
          fbSet(fb, "hint", "💡 Not the same size. Look at the red bars. Do they line up?");
        }
      };
    });
  }

  const n = Progress.badges.size, next = MISSIONS.find(m => !Progress.has(m.id));
  talk($("#hqTalk"), "cap", n === 0
    ? "Count Crumble broke Fraction City into pieces! Help us fix it. Pick a mission!"
    : n === 8 ? "You beat Count Crumble! You are a true Fraction Hero! Print your certificate."
      : `Great work! You have ${n} ${n === 1 ? "badge" : "badges"}. Try Mission ${next.id} next!`);
  $("#hqArt").innerHTML = figure("cap") + `<div style="transform:scale(.8);transform-origin:bottom">${figure("pint")}</div>` + `<div style="transform:scale(.72);transform-origin:bottom">${figure("boss")}</div>`;
  $("#hqCount").textContent = `${n} of 8 badges`; $("#hqBar").style.width = n / 8 * 100 + "%";
  const map = $("#map"); map.innerHTML = "";
  MISSIONS.forEach(m => {
    const done = Progress.has(m.id), isNext = next && next.id === m.id;
    const c = H("button", { class: `mcard${done ? " done" : ""}${isNext ? " next" : ""}${m.id === 8 ? " boss" : ""}`, "data-go": "m" + m.id, "aria-label": `Mission ${m.id}: ${m.title}. ${done ? "Badge earned." : "Not done yet."}` });
    c.innerHTML = `<span class="num">${m.id}</span>${isNext ? `<span class="ribbon">NEXT!</span>` : ""}${badgeSVG(m.id, done)}<span class="t">${m.title}</span><span class="s">${m.sub}</span><span class="s">${done ? "✅ Done! Play again" : "▶ Play"}</span>`;
    map.appendChild(c);
  });
}

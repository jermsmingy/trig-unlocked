  return { ok: false, valid: true, msg: "Valid electron count and octets, but the formal charges are on the wrong atoms. Negative formal charge belongs on the more electronegative atom." };
}

/* fractions for resonance averages */
function frac(x) {
  if (Math.abs(x - Math.round(x)) < 1e-9) return String(Math.round(x)).replace("-", "−");
  for (let d = 2; d <= 6; d++) { const n = Math.round(x * d); if (Math.abs(n / d - x) < 1e-9) return (n < 0 ? "−" : "") + Math.abs(n) + "/" + d; }
  return x.toFixed(2);
}

const _slotCache = {};
function angDist(a, b) { let d = Math.abs(a - b) % 360; return d > 180 ? 360 - d : d; }
function bestSlots(bondAngles, k) {
  if (k <= 0) return [];
  const key = bondAngles.map(a => Math.round(a)).sort((a, b) => a - b).join(",") + "|" + k;
  if (_slotCache[key]) return _slotCache[key];
  const slots = []; for (let a = 0; a < 360; a += 15) slots.push(a);
  let best = null, bestScore = -1;
  const pick = (start, chosen) => {
    if (chosen.length === k) {
      const all = bondAngles.concat(chosen); let mn = 999, sum = 0;
      for (let i = 0; i < all.length; i++) for (let j = i + 1; j < all.length; j++) { const d = angDist(all[i], all[j]); mn = Math.min(mn, d); if (i >= bondAngles.length || j >= bondAngles.length) sum += d; }
      const card = chosen.filter(a => a % 90 === 0).length;
      const score = mn * 1e6 + card * 1e3 + sum;
      if (score > bestScore) { bestScore = score; best = chosen.slice(); }
      return;
    }
    for (let i = start; i < slots.length; i++) { chosen.push(slots[i]); pick(i + 1, chosen); chosen.pop(); }
  };
  pick(0, []);
  return (_slotCache[key] = best);
}

/* ---------- valence electron counter (safe formula parser, no eval) ---------- */
function parseFormula(str) {
  const s = str.replace(/\s+/g, ""); if (!s) throw new Error("Type a formula like SO4 or CH3COOH");
  if (!/^[A-Za-z0-9()]+$/.test(s)) throw new Error("Use element symbols, numbers, and parentheses only");
  let i = 0;
  function group() {
    const counts = {};
    while (i < s.length && s[i] !== ")") {
      let part;
      if (s[i] === "(") { i++; part = group(); if (s[i] !== ")") throw new Error("Missing )"); i++; }
      else { const m = s.slice(i).match(/^[A-Z][a-z]?/); if (!m) throw new Error(`"${s[i]}" isn't an element symbol (symbols start with a capital letter)`);
        const sym = EL[m[0]] ? m[0] : (EL[m[0][0]] && m[0].length === 2 && !EL[m[0]] ? null : m[0]);
        if (sym === null || !EL[sym]) throw new Error(`"${m[0]}" isn't a main-group element in this lesson`);
        i += sym.length; part = { [sym]: 1 }; }
      const n = s.slice(i).match(/^\d+/); const k = n ? parseInt(n[0], 10) : 1; if (n) i += n[0].length;
      if (k < 1 || k > 99) throw new Error("Subscripts must be between 1 and 99");
      for (const e in part) counts[e] = (counts[e] || 0) + part[e] * k;
    }
    return counts;
  }
  const c = group(); if (i < s.length) throw new Error("Unmatched )"); return c;
}

  sel.innerHTML = "";
  const cats = [...new Set(MOLS.map(m => m.cat))];
  cats.forEach(c => {
    const og = document.createElement("optgroup"); og.label = c;
    MOLS.filter(m => m.cat === c && filter(m)).forEach(m => { const o = document.createElement("option"); o.value = m.id; o.textContent = `${solved && solved.has(m.id) ? function centralText(m) {
  const cen = centralAtoms(m);
  if (m.atoms.length === 2) return "Diatomic: just connect the two atoms.";
  if (cen.length > 1) return `More than one "central" atom here (${cen.map(i => m.atoms[i]).join(", ")}). The skeleton follows the way the formula is written.`;
  const c = m.atoms[cen[0]], others = [...new Set(m.atoms.filter((a, i) => i !== cen[0]))];
  const why = others.includes("H") && others.length === 1 ? "H can only form one bond, so it's never central" : `${c} is the least electronegative (${EL[c].en}) of the non-H atoms${others.includes("H") ? "; H is never central" : ""}`;
  return `Central atom: <b>${c}</b>. ${why}.`;
}
function breakdownText(m) {
  const b = valenceBreakdown(m);
  return b.parts.map(p => `${p.el}: ${p.n} × ${p.ve} = ${p.sub}`).join(" · ") + (m.q ? ` · charge ${Math.abs(m.q)}${m.q < 0 ? "−" : "+"} → ${m.q < 0 ? "add" : "subtract"} ${Math.abs(m.q)}` : "") + ` · <b>total ${b.total}</b>`;
}

/* grading for typed answers (exported for testing) */
function gradeInput(P, v) {
  const close = (a, b) => Math.abs(a - b) < 1e-6;
  if (P.type === "in") {
    if (/^\s*(und|dne|undefined|none)/i.test(v)) return { ok: false, msg: "It's actually defined here." };
    let n; try { n = makeFn(v, { noVar: true }).f(0); } catch (e) { return { nudge: "⚠ " + escapeHTML(e.message) }; }
    if (!isFinite(n)) return { ok: false, msg: "That expression is undefined." };
    if (/\d\.\d|^\.\d|[^\d]\.\d/.test(v)) return close(n, P.ans) || Math.abs(n - P.ans) < 0.0006 ? { nudge: "🔢 That's the right <i>decimal</i>! Now write it as an <b>exact</b> value (use √ and fractions)." } : { ok: false };
    if (close(n, P.ans)) return { ok: true };
    if (close(n, -P.ans)) return { ok: false, msg: "Right size, wrong <b>sign</b>. Check the quadrant!" };
    if (close(n, 1 / P.ans)) return { ok: false, msg: "That's the <b>reciprocal</b> of the answer." };
    return { ok: false };
  }
  if (P.type === "list") {
    let vals;
    try { vals = v.split(/[,;]|\band\b/).map(s => s.trim()).filter(Boolean).map(s => makeFn(s, { noVar: true }).f(0)); }
    catch (e) { return { nudge: "⚠ " + escapeHTML(e.message) }; }
    const asDeg = vals.every(x => P.ans.some(a => close(x * PI / 180, a))) && vals.some(x => x > 7);
    if (asDeg) return { nudge: "Those look like <b>degrees</b>. The interval [0, 2π) means radians, e.g. <code>pi/3</code>." };
    if (/\d\.\d/.test(v)) return { nudge: "🔢 Use exact values with π, like <code>2pi/3</code>, not decimals." };
    const outside = vals.filter(x => x < -1e-9 || x >= TAU - 1e-9);
    if (outside.length) return { ok: false, msg: "Some answers are outside [0, 2π)." };
    const found = P.ans.filter(a => vals.some(x => close(x, a))).length;
    const extra = vals.filter(x => !P.ans.some(a => close(x, a))).length;
    if (found === P.ans.length && !extra) return { ok: true };
    if (!extra && found > 0) return { ok: false, msg: `You found ${found} of ${P.ans.length} solutions. There are more!` };
    return { ok: false, msg: extra ? `${extra} of your answers ${extra > 1 ? "don't" : "doesn't"} work.` : "" };
  }
}

    const d = SPECIAL[Math.floor(Math.random() * 16)], f = ["sin", "cos", "tan"][Math.floor(Math.random() * 3)], ex = exactTrig(d);

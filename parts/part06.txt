function gradeNum(P, v) {
  const t = v.trim().replace(/[−–]/g, "-").replace(/\s+/g, "");
  if (!/^[+-]?\d+$/.test(t)) return { nudge: "Enter a whole number, like 24, +1, or -1." };
  const n = parseInt(t, 10);
  if (n === P.ans) return { ok: true };
  if (n === -P.ans && n !== 0) return { ok: false, msg: "Check the <b>sign</b>." };
  return { ok: false };
}
